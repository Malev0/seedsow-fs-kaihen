# Seedsow FS exploit - run configuration.
#
# These are the working values; the GUI and the CLI both read this file. A
# machine can override any of them in env.local.txt (not tracked) - that file is
# read second and wins. Device-specific settings do not belong here:
#   console address            -> tools/console-host.txt
#   controller address + token -> controller-connection.json

# --- Stage 2: FS memory acquisition ---------------------------------------
# Alias buckets tried per boot, best first. Buckets that cannot alias on this
# console (0x1500000, 0x15A0000, 0x12B0000) are deliberately absent.
NS2_ALIAS_OFFSETS=0x14A0000,0x13B0000,0x1450000,0x1400000,0x1360000
NS2_ALIAS_BUCKETS=3
# Positive-canary waves before a boot is called a miss.
NS2_POSITIVE_ATTEMPTS=8
# Allocate/free rounds before the link search (1 = the original behaviour).
NS2_GROOM_ROUNDS=3
# Wave caps per phase.
NS2_ALIAS_WAVE_CAP=21
NS2_ADDRESS_WAVE_CAP=20
NS2_POSITIVE_WAVE_MS=10000
# Cache read size and the 8 MiB dirty commit that lets the canary interleave.
NS2_CACHE_READ_SIZE=0x200000
NS2_COMMIT_MB=8
# Reject the low ~4 GiB app/static mapping targets before planning.
NS2_MIN_TARGET=0x800000000
# Stable-only candidates wedge the FS at the address read: fail cleanly.
NS2_STABLE_ATTEMPT=0

# --- Stage 3: retained endpoint and RO launch ------------------------------
NS2_CONNECT_OFFSET=0x168C34
NS2_GATE_WAKE=park
NS2_GATE_WAKE_MS=25
NS2_CANCEL_SETTLE_MS=200
NS2_GATE_STEP_MODE=inject
NS2_GATE_WORKER=1
NS2_GATE_STEP_MS=50
NS2_GATE_SERVE_MS=250
NS2_GATE_IDLE_REBUILD=1
NS2_GATE_UNPARK=0
NS2_GATE_UNPARK_SETTLE_MS=300
NS2_GATE_UNPARK_ACCEPT=1
NS2_GATE_REBUILD_HOLDER=0
NS2_PARK_SETTLE_MS=1500
NS2_SKIP_TEST_GATE=1
# The Loader mounts the code FS before the start and again after it; the second
# mount is the one its /main read uses, so it is exchanged (mount 2, two-way).
NS2_LAUNCH_MOUNT=2
NS2_MAIN_SWAP=1
NS2_MAIN_SWAP_MODE=capture
# 0: the pre-start exchange above is enough. A second post-start swap leaves the
# code mount in a state the app cannot be torn down from.
NS2_POST_START_OPEN=0
NS2_POST_START_SWAP=mount
NS2_OPEN_SWAP=0
NS2_CODE_SWAP=0
NS2_FS_SWAP=0
NS2_EXEFS_REDIRECT=0
# Terminate the stock RO before starting the replacement.
NS2_STOCK_TERMINATE=pre
# flags 0x10 = StartSuspended: the mount exchange happens before the code runs.
NS2_LAUNCH_FLAGS=0x10
NS2_STOP_AFTER_LAUNCH=0
NS2_EXCEPTION_CHECK=0

# --- Stage 4: kaiHEN startup ----------------------------------------------
# 1 = run kaiHEN's HEN after the pipeline (the normal flow).
NS2_HEN=1
# 0 = the bootstrap writes and opens /kaihen/bin/hen/exefs.nsp itself (upstream).
# 1 = the host builds, stages and opens it and hands the bootstrap the row.
# stage = the host builds and stages it but supplies no row; the bootstrap still
#         writes and opens it (the host's staged copy is replaced).
# 0 is the current test: the kit's bootstrap polls the gate instead of calling
# TryWaitThread, which is what wedged the upstream package flow.
# 1 requires the host-row bootstrap at commit 55a6f57; this stock-flow
# bootstrap ignores the row and would also write the package (0xE02).
NS2_HEN_HOST_PACKAGE=0
# 0 = arm R-held launches as usual (the point of the port).
NS2_SKIP_GAME_ARM=0
# Retire the exploit endpoint after Complete, restoring the poisoned cache entry
# that FS cannot finalise. "delayed" waits NS2_HEN_RETIRE_MS so HEN has done its
# one retained-alias read first.
NS2_HEN_RETIRE=delayed
NS2_HEN_RETIRE_MS=8000
# Restore the entry and skip our own teardown: the app's close does it safely.
NS2_RETIRE_LEAK=1
# Stop the console-side RPC loop before the app is closed, otherwise the app
# never exits ("closing software" spins forever).
NS2_HEN_DETACH=1
NS2_HEN_DETACH_DELAY_MS=0
# Leave the bootstrap module mapped in the app.
NS2_HEN_KEEP_MAPPED=1
