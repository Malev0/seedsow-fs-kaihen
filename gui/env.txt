# Extra pipeline overrides, read on every Start (blank lines and # are ignored).
# One KEY=VALUE per line. Examples to try, one change at a time:
#
#   NS2_SLEEP_RAW=1            # undo the 19.2/31.25 MHz sleep scaling
#   NS2_COMMIT_MB=8            # shorter dirty commit; lets the canary interleave
#   NS2_SKIP_REPRO=1           # accept one confirmed marker (no second positive)
#   NS2_SKIP_EXTENT_REQUAL=1   # keep the proven extent after each marker write
#   NS2_WORKER_PRIO=5          # worker priority (5..35, 5 = highest accepted)
#   NS2_WORKER_AFFINITY=1      # pin workers at creation, core 0 excluded
#   NS2_WAVE_SETTLE_MS=0       # no per-retry settle (faster, less reliable)
#   NS2_GATE_WAKE=park         # gate wake target: park (default), both, receive, 0
#   NS2_GATE_WAKE_MS=25        # settle between waking and clearing the event
#
# FS module image probe (read-only, runs after prepare-workers, before the gate).
# NOTE: the retained alias cannot leave the FS heap. Pointing it at FS text makes
# the candidate file read block for good (endpoint wedged, console still alive),
# so leave these off unless that mechanism changes.
#   NS2_FS_PROBE=1             # read the API-thread entry
#   NS2_FS_DUMP_BYTES=2621440  # optional: dump 2.5 MiB from the module base
#   NS2_FS_PROBE_ONLY=1        # stop before the gate (the gate hard-hangs the console)
#
# The run log prints "[gui] env overrides: ..." so you can confirm what applied.
NS2_GATE_WAKE=park
# Holder test: run the inject path with one real control-test request. The FS
# writes row.address - 0x88 into the waiter's selected slot on a natural wake,
# so inject-holder now stores that holder pointer (validated in memory) instead
# of the session row. NS2_GATE_STEP_MODE=inject stops test-gate after the first
# injected request so a wrong pointer cannot run the second step or the join.
NS2_GATE_STEP_MODE=inject
NS2_GATE_WORKER=1
NS2_GATE_UNPARK=0
NS2_GATE_STEP_MS=50
NS2_GATE_SERVE_MS=250
NS2_GATE_UNPARK_SETTLE_MS=300
NS2_GATE_UNPARK_ACCEPT=1
# A successful positive canary wave is fast; a slow one is a bad boot. Cap the
# whole positive retry series (and each join) at 10 s so stage 2 fails fast.
NS2_POSITIVE_WAVE_MS=10000
NS2_LAUNCH_MODE=gate
# flags 0x10 = StartSuspended. The launch creates the process but does no FS
# work; the mount is deferred to the pm:dmnt start. ro/launch now triggers that
# start inside the serve-gate window, so the mount can be observed and swapped.
# flags 0 loads inside the launch send, holds the FS server, and faults the app.
NS2_LAUNCH_FLAGS=0x10
NS2_LAUNCH_WAIT_ONLY=0
NS2_LAUNCH_WAIT_MS=90000
# The flags-0 load holds the launcher shim, so the host cannot read the slot-2
# worker's buffer. The worker sends the captured table window to this PC over
# TCP instead. Start `node tools/dump-listener.cjs` before pressing Start.
# Set NS2_DUMP_HOST to this PC's LAN address (the console connects back to it).
# Machine-specific values belong in gui/env.local.txt (gitignored); it overrides
# this file, so the repository keeps only this placeholder.
NS2_DUMP_HOST=192.168.1.50
NS2_DUMP_PORT=39401
# The stale second-session kick is off for this diagnostic.
NS2_LAUNCH_KICK_OTHER=0
# Keep the table baseline: the post-load diff is the point of the run.
NS2_LAUNCH_TABLE_SCAN=1
# The poll settings are unused in wait-only mode; keep them for the normal mode.
NS2_LAUNCH_POLL_MS=20
NS2_LAUNCH_DOMAIN_POLL_MS=1000
NS2_LAUNCH_TABLE_POLL_MS=5000
NS2_LAUNCH_WAKE_MS=1000
